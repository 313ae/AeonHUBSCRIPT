local BUTTON_FULL_SIZE = UDim2.new(0, 50, 0, 50)
local LOGO_FULL_SIZE = UDim2.new(0, 36, 0, 36)
local BUTTON_POS = UDim2.new(0, 35, 1, -90)

local function showToggleButton()
    if toggleButton.Visible then return end
    toggleButton.Visible = true
    TweenService:Create(toggleButton, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = BUTTON_FULL_SIZE,
        Position = BUTTON_POS
    }):Play()
    TweenService:Create(logo, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = LOGO_FULL_SIZE,
        TextSize = 34
    }):Play()
end

introFrame.Destroying:Connect(showToggleButton)
task.delay(10, function()
    if introFrame and introFrame.Parent then return end
    showToggleButton()
end)

do
    local dragging, dragInput, dragStart, startPos
    toggleButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = toggleButton.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    toggleButton.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            local delta = input.Position - dragStart
            toggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local FULL_SIZE = UDim2.new(0, 460, 0, 330)
local SAVED_POS = UDim2.new(0.5, -230, 0.5, -165)
do
    local clickStart
    local pressStartTime
    local guiVisible = false
    local SMALL_SIZE = UDim2.new(0, 0, 0, 0)

    local function buttonCenter()
        return UDim2.new(
            toggleButton.Position.X.Scale, toggleButton.Position.X.Offset + 25,
            toggleButton.Position.Y.Scale, toggleButton.Position.Y.Offset + 25
        )
    end

    toggleButton.MouseButton1Down:Connect(function()
        clickStart = { x = toggleButton.Position.X.Offset, y = toggleButton.Position.Y.Offset }
        pressStartTime = os.clock()
    end)

    toggleButton.MouseButton1Up:Connect(function()
        if not clickStart then return end
        local pos = toggleButton.Position
        local moved = math.abs(pos.X.Offset - clickStart.x) + math.abs(pos.Y.Offset - clickStart.y)
        local quickPress = pressStartTime and os.clock() - pressStartTime < 0.25
        clickStart = nil
        pressStartTime = nil
        if quickPress then
            local p = toggleButton.Position
            local shrink = TweenService:Create(toggleButton, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 44, 0, 44),
                Position = UDim2.new(p.X.Scale, p.X.Offset + 3, p.Y.Scale, p.Y.Offset + 3)
            })
            shrink:Play()
            TweenService:Create(logo, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 30, 0, 30),
                TextSize = 26
            }):Play()
            shrink.Completed:Connect(function()
                local p2 = toggleButton.Position
                TweenService:Create(toggleButton, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = BUTTON_FULL_SIZE,
                    Position = UDim2.new(p2.X.Scale, p2.X.Offset - 3, p2.Y.Scale, p2.Y.Offset - 3)
                }):Play()
                TweenService:Create(logo, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = LOGO_FULL_SIZE,
                    TextSize = 34
                }):Play()
            end)
        end
        if moved >= 10 then return end

        guiVisible = not guiVisible
        if guiVisible then
            mainFrame.Visible = true
            mainFrame.Size = SMALL_SIZE
            mainFrame.Position = buttonCenter()
            local openTween = TweenService:Create(mainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = FULL_SIZE,
                Position = SAVED_POS
            })
            openTween:Play()
        else
            local closeTween = TweenService:Create(mainFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
                Size = SMALL_SIZE,
                Position = buttonCenter()
            })
            closeTween:Play()
            closeTween.Completed:Connect(function()
                if not guiVisible then
                    mainFrame.Visible = false
                end
            end)
        end
    end)
end

	local function refreshSlider()
		local ratio = (speedValue - MIN) / (MAX - MIN)
		sliderFill.Size = UDim2.new(0, ratio * SPEED_TRACK_W, 1, 0)
		sliderKnob.Position = UDim2.new(0, ratio * SPEED_TRACK_W, 0.5, 0)
		sliderValueLabel.Text = tostring(math.floor(speedValue + 0.5)) .. "/" .. MAX
	end
	refreshSlider()

		local ratio = math.clamp((inputPos.X - sliderTrack.AbsolutePosition.X) / SPEED_TRACK_W, 0, 1)
		speedValue = MIN + ratio * (MAX - MIN)
		refreshSlider()

	local function stopFly()
		if flyConn then flyConn:Disconnect() flyConn = nil end

		if flyGyro then pcall(function() flyGyro:Destroy() end) flyGyro = nil end

	local function startFly()
		local char = player.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not hrp or not hum then return end
		hum.PlatformStand = true
		flyVel = Instance.new("BodyVelocity")
		flyVel.MaxForce = Vector3.new(9e9, 9e9, 9e9)
		flyVel.Velocity = Vector3.new(0, 0, 0)
		flyVel.Parent = hrp

		flyGyro.Parent = hrp

			if not flyOn then return end
			local cam = workspace.CurrentCamera
			if not cam or not hrp.Parent then return end
			local move = Vector3.new(0, 0, 0)
			if flyKeys[Enum.KeyCode.W] then move = move + cam.CFrame.LookVector end
			if flyKeys[Enum.KeyCode.S] then move = move - cam.CFrame.LookVector end
			if flyKeys[Enum.KeyCode.D] then move = move + cam.CFrame.RightVector end
			if flyKeys[Enum.KeyCode.A] then move = move - cam.CFrame.RightVector end
			if flyKeys[Enum.KeyCode.Space] or flyKeys[Enum.KeyCode.E] then move = move + Vector3.new(0, 1, 0) end
			if flyKeys[Enum.KeyCode.LeftShift] or flyKeys[Enum.KeyCode.Q] then move = move - Vector3.new(0, 1, 0) end
			if move.Magnitude > 0 then move = move.Unit * flyValue else move = Vector3.new(0, 0, 0) end
			flyVel.Velocity = move
			flyGyro.CFrame = cam.CFrame
		end)
	end

		if flyOn then stopFly() startFly() end

	-- Noclip
	local noclipOn = false
	local noclipConn = nil
	local noclipSaved = {}
	local function applyNoclip(char)
		if not char then return end
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then
				if noclipSaved[part] == nil then noclipSaved[part] = part.CanCollide end
				part.CanCollide = false
			end
		end
	end
	local function restoreNoclip()
		for part, was in pairs(noclipSaved) do
			if part and part.Parent then
				part.CanCollide = was
			end
		end
		noclipSaved = {}
	end
	local noclipTrack, noclipKnob
	local function setNoclipOn(state)
		noclipOn = state
		if noclipConn then noclipConn:Disconnect() noclipConn = nil end
		if state then
			noclipTrack.BackgroundColor3 = LIME
			noclipKnob.Position = UDim2.new(1, -18, 0, 2)
			noclipKnob.BackgroundColor3 = BG_DARKER
			local char = player.Character
			if char then
				noclipSaved = {}
				applyNoclip(char)
				noclipConn = RunService.Stepped:Connect(function() applyNoclip(char) end)
			end
		else
			noclipTrack.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
			noclipKnob.Position = UDim2.new(0, 2, 0, 2)
			noclipKnob.BackgroundColor3 = TEXT_SECONDARY
			restoreNoclip()
		end
	end
	player.CharacterAdded:Connect(function(char)
		noclipSaved = {}
		if noclipOn then
			task.wait(0.5)
			if player.Character == char then
				applyNoclip(char)
				if noclipConn then noclipConn:Disconnect() end
				noclipConn = RunService.Stepped:Connect(function() applyNoclip(char) end)
			end
		end
	end)

local activeIndicator = Instance.new("Frame")
activeIndicator.Name = "ActiveIndicator"
activeIndicator.Size = UDim2.new(0, 2, 0, 40)
activeIndicator.Position = UDim2.new(0, 0, 0, 0)

activeIndicator.BorderSizePixel = 0
activeIndicator.ZIndex = 3
activeIndicator.Parent = body

local tabIndex = { Main = 1, Dashboard = 2, Misc = 3, Settings = 4 }

local function setActiveTab(name, animate)
	if animate == nil then animate = true end
	for tabName, button in pairs(tabButtons) do
		local label = button:FindFirstChild("Label")
		if tabName == name then
			label.TextColor3 = TEXT_PRIMARY
			label.Font = Enum.Font.GothamBold
		else
			label.TextColor3 = TEXT_SECONDARY
			label.Font = Enum.Font.GothamBold
		end
	end

	local idx = tabIndex[name] or 1
	local targetPos = UDim2.new(0, 0, 0, (idx - 1) * 40)
	if animate then
		TweenService:Create(activeIndicator, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			Position = targetPos
		}):Play()
	else
		activeIndicator.Position = targetPos
	end

	for pageName, page in pairs(pages) do
		page.Visible = (pageName == name)
	end
end
