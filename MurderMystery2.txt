do

    toggleButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = toggleButton.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    toggleButton.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input

        if dragging and input == dragInput then
            local delta = input.Position - dragStart
            toggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

local FULL_SIZE = UDim2.new(0, 460, 0, 330)
local SAVED_POS = UDim2.new(0.5, -230, 0.5, -165)
do
    local clickStart
    local pressStartTime
    local guiVisible = false
    local SMALL_SIZE = UDim2.new(0, 0, 0, 0)

    local function buttonCenter()
        return UDim2.new(
            toggleButton.Position.X.Scale, toggleButton.Position.X.Offset + 25,
            toggleButton.Position.Y.Scale, toggleButton.Position.Y.Offset + 25
        )
    end

    toggleButton.MouseButton1Down:Connect(function()
        clickStart = { x = toggleButton.Position.X.Offset, y = toggleButton.Position.Y.Offset }
        pressStartTime = os.clock()
    end)

    toggleButton.MouseButton1Up:Connect(function()
        if not clickStart then return end
        local pos = toggleButton.Position
        local moved = math.abs(pos.X.Offset - clickStart.x) + math.abs(pos.Y.Offset - clickStart.y)
        local quickPress = pressStartTime and os.clock() - pressStartTime < 0.25
        clickStart = nil
        pressStartTime = nil
        if quickPress then
            local p = toggleButton.Position
            local shrink = TweenService:Create(toggleButton, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 44, 0, 44),
                Position = UDim2.new(p.X.Scale, p.X.Offset + 3, p.Y.Scale, p.Y.Offset + 3)
            })
            shrink:Play()
            TweenService:Create(logo, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 30, 0, 30),
                TextSize = 26
            }):Play()
            shrink.Completed:Connect(function()
                local p2 = toggleButton.Position
                TweenService:Create(toggleButton, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = BUTTON_FULL_SIZE,
                    Position = UDim2.new(p2.X.Scale, p2.X.Offset - 3, p2.Y.Scale, p2.Y.Offset - 3)
                }):Play()
                TweenService:Create(logo, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = LOGO_FULL_SIZE,
                    TextSize = 34
                }):Play()
            end)
        end
        if moved >= 10 then return end

        guiVisible = not guiVisible
        if guiVisible then
            mainFrame.Visible = true
            mainFrame.Size = SMALL_SIZE
            mainFrame.Position = buttonCenter()
            local openTween = TweenService:Create(mainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = FULL_SIZE,
                Position = SAVED_POS
            })
            openTween:Play()
        else
            local closeTween = TweenService:Create(mainFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
                Size = SMALL_SIZE,
                Position = buttonCenter()
            })
            closeTween:Play()
            closeTween.Completed:Connect(function()
                if not guiVisible then
                    mainFrame.Visible = false
                end
            end)
        end
    end)
end

	local innocentESPEnabled = false
	local innocentHighlights = {}
	local function innocentHasTool(p, name)
		local char = p.Character
		if char and char:FindFirstChild(name) then return true end
		local bp = p:FindFirstChild("Backpack")
		if bp and bp:FindFirstChild(name) then return true end
		for _, v in ipairs(char and char:GetDescendants() or {}) do
			if v.Name:lower():find(name:lower()) and (v:IsA("Tool") or v:IsA("BasePart")) then return true end
		end
		for _, v in ipairs(bp and bp:GetDescendants() or {}) do
			if v.Name:lower():find(name:lower()) and (v:IsA("Tool") or v:IsA("BasePart")) then return true end
		end
		return false
	end
	local function innocentDirectTool(p)
		local char = p.Character
		if char then
			if char:FindFirstChild("Knife") or char:FindFirstChild("Gun") or char:FindFirstChild("Revolver") then return true end
		end
		local bp = p:FindFirstChild("Backpack")
		if bp then
			if bp:FindFirstChild("Knife") or bp:FindFirstChild("Gun") or bp:FindFirstChild("Revolver") then return true end
		end
		return false
	end
	local function innocentIsDead(p)
		return false
	end
	local function hasExactTool(p, ...)
		local names = { ... }
		local char = p.Character
		if char then
			for _, c in ipairs(char:GetChildren()) do
				for _, n in ipairs(names) do
					if c.Name == n then return true end
				end
			end
		end
		local bp = p:FindFirstChild("Backpack")
		if bp then
			for _, c in ipairs(bp:GetChildren()) do
				for _, n in ipairs(names) do
					if c.Name == n then return true end
				end
			end
		end
		return false
	end
	local function hasKnife(p)
		return hasExactTool(p, "Knife")
	end
	local function hasGun(p)
		return hasExactTool(p, "Gun", "Revolver")
	end
	local roundMod = nil
	local function serverRole(name)
		if not roundMod then
			pcall(function()
				for _, d in ipairs(game:GetService("ReplicatedStorage"):GetDescendants()) do
					if d.Name == "CurrentRoundClient" and d:IsA("ModuleScript") then
						roundMod = d
						break
					end
				end
			end)
		end
		if roundMod then
			local ok, role = pcall(function()
				local tbl = require(roundMod)
				if type(tbl) == "table" and type(tbl.PlayerData) == "table" then
					local e = tbl.PlayerData[name]
					if type(e) == "table" and type(e.Role) == "string" then return e.Role end
				end
				return nil
			end)
			if ok then return role end
		end
		return nil
	end
	local function detectRole(p)
		local r = serverRole(p.Name)
		if r == "Murderer" or r == "Sheriff" or r == "Hero" then
			return r == "Hero" and "Sheriff" or r
		end
		if r ~= nil then return nil end
		if hasKnife(p) then return "Murderer" end
		if hasGun(p) then return "Sheriff" end
		return nil
	end
	local function innocentRole(p)
		if detectRole(p) then return "Armed" end
		return nil
	end
	local function innocentAdd(p)
		if not alive() then return end
		if p == player then return end
		if innocentIsDead(p) then return end
		if innocentHasTool(p, "Knife") or innocentHasTool(p, "Gun") or innocentHasTool(p, "Revolver") then return end
		if innocentRole(p) then return end
		local char = p.Character
		if not char or not char.Parent then return end
		if not char:FindFirstChild("HumanoidRootPart") and not char:FindFirstChild("UpperTorso") and not char:FindFirstChild("Torso") and not char:FindFirstChildWhichIsA("BasePart", true) then return end
		local existing = char:FindFirstChild("InnocentESP_" .. MY_GEN)
		if existing and existing:IsA("Highlight") then innocentHighlights[p] = existing return end
		local known = innocentHighlights[p]
		if known and known.Parent then return end
		innocentHighlights[p] = nil
		local hl = Instance.new("Highlight")
		hl.Name = "InnocentESP_" .. MY_GEN
		hl.FillTransparency = 1
		hl.OutlineTransparency = 0
		hl.OutlineColor = Color3.fromRGB(50, 210, 70)
		hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		hl.Adornee = char
		local okParent = pcall(function() hl.Parent = char end)
		if not okParent or hl.Parent ~= char then pcall(function() hl:Destroy() end) return end
		innocentHighlights[p] = hl
	end
	local function innocentRemove(p)
		local hl = innocentHighlights[p]
		if hl then pcall(function() hl:Destroy() end) innocentHighlights[p] = nil end
		local char = p.Character
		if char then
			for _, d in ipairs(char:GetDescendants()) do
				if d:IsA("Highlight") and (d.Name == "InnocentESP_" .. MY_GEN or d.Name == "InnocentESP") then
					pcall(function() d:Destroy() end)
				end
			end
		end
	end
	local function innocentClear()
		local ps = {}
		for p, _ in pairs(innocentHighlights) do ps[#ps + 1] = p end
		for _, p in ipairs(ps) do innocentRemove(p) end
	end
	local syncing = {}
	local function innocentSync(p)
		if syncing[p] then return end
		syncing[p] = true
		local ok, err = pcall(function()
			local hl = innocentHighlights[p]
			if hl and hl.Parent then
				if innocentESPEnabled and not innocentIsDead(p) and not innocentDirectTool(p) and not innocentRole(p) then
					return
				end
				innocentRemove(p)
			else
				if hl then innocentHighlights[p] = nil end
				innocentRemove(p)
			end
			if innocentESPEnabled then innocentAdd(p) end
		end)
		syncing[p] = nil
		if not ok then error(err, 0) end
	end

							boxWidth, screenSize = math.floor(boxWidth + 0.5), math.floor(screenSize + 0.5)
							box.Size = Vector2.new(boxWidth, screenSize)
							box.Position = Vector2.new(px, py)
							box.Color = boxColor(p)

						end
					end
				end
			end

	-- Noclip
	local noclipOn = false
	local noclipConn = nil
	local noclipSaved = {}
	local function applyNoclip(char)
		if not alive() then return end
		if not char then return end
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then
				if noclipSaved[part] == nil then noclipSaved[part] = part.CanCollide end
				part.CanCollide = false
			end
		end
	end
	local function restoreNoclip()
		for part, was in pairs(noclipSaved) do
			if part and part.Parent then
				part.CanCollide = was
			end
		end
		noclipSaved = {}
	end
	local noclipTrack, noclipKnob
	local function setNoclipOn(state)
		noclipOn = state
		if noclipConn then noclipConn:Disconnect() noclipConn = nil end
		if state then
			TweenService:Create(noclipTrack, TweenInfo.new(0.15, Enum.EasingStyle.Quad), { BackgroundColor3 = LIME }):Play()
			TweenService:Create(noclipKnob, TweenInfo.new(0.15, Enum.EasingStyle.Quad), { Position = UDim2.new(1, -18, 0, 2), BackgroundColor3 = BG_DARKER }):Play()
			local char = player.Character
			if char then
				noclipSaved = {}
				applyNoclip(char)
				noclipConn = RunService.Stepped:Connect(function() applyNoclip(char) end)
			end
		else
			TweenService:Create(noclipTrack, TweenInfo.new(0.15, Enum.EasingStyle.Quad), { BackgroundColor3 = Color3.fromRGB(60, 60, 60) }):Play()
			TweenService:Create(noclipKnob, TweenInfo.new(0.15, Enum.EasingStyle.Quad), { Position = UDim2.new(0, 2, 0, 2), BackgroundColor3 = TEXT_SECONDARY }):Play()
			restoreNoclip()
		end
	end
	player.CharacterAdded:Connect(function(char)
		noclipSaved = {}
		if noclipOn then
			task.wait(0.5)
			if player.Character == char then
				applyNoclip(char)
				if noclipConn then noclipConn:Disconnect() end
				noclipConn = RunService.Stepped:Connect(function() applyNoclip(char) end)
			end
		end
	end)
