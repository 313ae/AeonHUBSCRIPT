local function sha256hex(msg)
    local K = {
        0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5, 0x3956c25b, 0x59f111f1, 0x923f82a4, 0xab1c5ed5,

        0x27b70a85, 0x2e1b2138, 0x4d2c6dfc, 0x53380d13, 0x650a7354, 0x766a0abb, 0x81c2c92e, 0x92722c85,
        0xa2bfe8a1, 0xa81a664b, 0xc24b8b70, 0xc76c51a3, 0xd192e819, 0xd6990624, 0xf40e3585, 0x106aa070,
        0x19a4c116, 0x1e376c08, 0x2748774c, 0x34b0bcb5, 0x391c0cb3, 0x4ed8aa4a, 0x5b9cca4f, 0x682e6ff3,
        0x748f82ee, 0x78a5636f, 0x84c87814, 0x8cc70208, 0x90befffa, 0xa4506ceb, 0xbef9a3f7, 0xc67178f2,
    }
    local h0, h1, h2, h3, h4, h5, h6, h7 =
        0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19
    local ml = #msg * 8
    msg = msg .. string.char(0x80)
    msg = msg .. string.rep(string.char(0), (56 - (#msg % 64)) % 64)
    local len = ""
    for i = 7, 0, -1 do
        len = len .. string.char(math.floor(ml / (256 ^ i)) % 256)
    end
    msg = msg .. len
    local w = {}
    for chunk = 1, #msg, 64 do
        for i = 0, 15 do
            local o = chunk + i * 4
            local b1, b2, b3, b4 = string.byte(msg, o, o + 3)
            w[i + 1] = ((b1 * 256 + b2) * 256 + b3) * 256 + b4
        end
        for j = 17, 64 do
            local x, y = w[j - 15], w[j - 2]
            local s0 = bit32.bxor(bit32.bxor(bit32.rrotate(x, 7), bit32.rrotate(x, 18)), bit32.rshift(x, 3))
            local s1 = bit32.bxor(bit32.bxor(bit32.rrotate(y, 17), bit32.rrotate(y, 19)), bit32.rshift(y, 10))
            w[j] = (w[j - 16] + s0 + w[j - 7] + s1) % 4294967296
        end
        local a, b, c, d, e, f, g, h = h0, h1, h2, h3, h4, h5, h6, h7
        for j = 1, 64 do
            local S1 = bit32.bxor(bit32.bxor(bit32.rrotate(e, 6), bit32.rrotate(e, 11)), bit32.rrotate(e, 25))
            local ch = bit32.bxor(bit32.band(e, f), bit32.band(bit32.bnot(e), g))
            local t1 = (h + S1 + ch + K[j] + w[j]) % 4294967296
            local S0 = bit32.bxor(bit32.bxor(bit32.rrotate(a, 2), bit32.rrotate(a, 13)), bit32.rrotate(a, 22))
            local maj = bit32.bxor(bit32.bxor(bit32.band(a, b), bit32.band(a, c)), bit32.band(b, c))
            local t2 = (S0 + maj) % 4294967296
            h = g; g = f; f = e; e = (d + t1) % 4294967296; d = c; c = b; b = a; a = (t1 + t2) % 4294967296
        end
        h0 = (h0 + a) % 4294967296; h1 = (h1 + b) % 4294967296
        h2 = (h2 + c) % 4294967296; h3 = (h3 + d) % 4294967296
        h4 = (h4 + e) % 4294967296; h5 = (h5 + f) % 4294967296
        h6 = (h6 + g) % 4294967296; h7 = (h7 + h) % 4294967296
    end
    return string.format("%08x%08x%08x%08x%08x%08x%08x%08x", h0, h1, h2, h3, h4, h5, h6, h7)
end

local function normKey(s)
    local t = tostring(s or ""):match("^%s*(.-)%s*$") or ""
    return t:upper()
end
local function getHWID()
    local ok, id = pcall(function() return gethwid() end)
    if ok and type(id) == "string" and #id > 0 then return id end
    ok, id = pcall(function() return get_hwid() end)
    if ok and type(id) == "string" and #id > 0 then return id end
    return nil
end
local function fetchKeys()
    local ok, body = pcall(function()
        return game:HttpGet(KEY_URL .. "?t=" .. tostring(os.time()))
    end)
    if not ok or type(body) ~= "string" then return nil end
    local ok2, data = pcall(function() return HttpService:JSONDecode(body) end)
    if ok2 and type(data) == "table" then return data end
    return nil
end
local function postWebhook(url, content)
    local reqFn = nil
    pcall(function() reqFn = (syn and syn.request) or http_request or request end)
    if type(url) ~= "string" or #url == 0 or type(reqFn) ~= "function" then return end
    if type(content) ~= "string" or #content == 0 then return end
    pcall(function()
        reqFn({
            Url = url,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = HttpService:JSONEncode({ content = content }),
        })
    end)
end

local function checkKey(key)
    if #key < 8 then return "bad", nil end
    local db = fetchKeys()
    if not db then return "netfail", nil end
    local h = sha256hex(key)
    local rec = db[h]
    if type(rec) ~= "table" then return "bad", nil end
    if rec.banned then return "banned", rec end
    if type(rec.expires) == "number" and rec.expires < os.time() then return "expired", rec end
    local hwid = getHWID()
    if hwid then
        local hwidHash = sha256hex(hwid)
        local stored = rec.hwidHash or rec.hwid
        if type(stored) == "string" and #stored > 0 then
            local cmp = stored
            if #stored ~= 64 then cmp = sha256hex(stored) end
            if cmp ~= hwidHash then return "hwid_mismatch", rec end
        end
    end
    return "ok", rec
end

local function selectTab(name, animate)
    local oldIndex, newIndex = 0, 0
    for i, n in ipairs(sectionNames) do
        if n == currentTab then oldIndex = i end
        if n == name then newIndex = i end
    end
    currentTab = name
    refreshBottomTabVisibility()
    if name == "Settings" and selectedBottomTab ~= "Features" then
        selectBottomTab("Features", false)
    end
    for panelName, panel in pairs(tabPanels) do
        panel.Visible = (panelName == name)
    end
    local shown = tabPanels[name]
    if shown then
        task.defer(function()
            for _, child in ipairs(shown:GetDescendants()) do
                if child:IsA("UIListLayout") then
                    local order = child.SortOrder
                    child.SortOrder = Enum.SortOrder.Name
                    child.SortOrder = order
                end
            end
        end)
        if animate and newIndex ~= oldIndex then
            local dir = (newIndex > oldIndex) and 1 or -1
            shown.Position = UDim2.new(0, 40 * dir, 0, 0)
            TweenService:Create(shown, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Position = UDim2.new(0, 0, 0, 0)
            }):Play()
        end
    end
    for btnName, btn in pairs(tabButtons) do
        btn.TextColor3 = (btnName == name) and ACTIVE_TEXT or MUTED_TEXT
    end

    local targetButton = tabButtons[name]
    local targetX = targetButton.AbsolutePosition.X - sectionBar.AbsolutePosition.X
        + (targetButton.AbsoluteSize.X - 20) / 2
    local targetPos = UDim2.new(0, targetX, 1, -2)

    if animate then
        TweenService:Create(activeUnderline, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Position = targetPos
        }):Play()
    else
        activeUnderline.Position = targetPos
    end
end

local function pushSample(history, value)
    table.insert(history, value)
    if #history > STAT_SAMPLES then
        table.remove(history, 1)
    end
end

local function redrawBars(bars, history)
    if #history == 0 then return end
    local minVal, maxVal = history[1], history[1]
    for _, v in ipairs(history) do
        minVal = math.min(minVal, v)
        maxVal = math.max(maxVal, v)
    end
    local range = math.max(maxVal - minVal, 1)
    for i, bar in ipairs(bars) do
        local sample = history[i]
        local h = 4
        if sample then
            h = 4 + ((sample - minVal) / range) * 12
        end
        if bar and bar.Parent then
            bar.Size = UDim2.new(0, 2, 0, h)
        end
    end
end

local function getPing()
    local ok, value = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    if ok and value then
        return math.floor(value + 0.5)
    end
    return 0
end

local frameCount, fpsElapsed, statElapsed = 0, 0, 0
local scrollbarThrottle = 0

trackLoop(RunService.Heartbeat:Connect(function(dt)
    if not mainFrame.Visible then
        frameCount, fpsElapsed, statElapsed = 0, 0, 0
        return
    end
    local now = os.clock()
    if now - scrollbarThrottle > 0.1 then
        scrollbarThrottle = now
        refreshScrollbar()
    end
    frameCount = frameCount + 1
    fpsElapsed = fpsElapsed + dt
    statElapsed = statElapsed + dt

    if statElapsed >= STAT_UPDATE_INTERVAL then
        local fps = fpsElapsed > 0 and math.floor((frameCount / fpsElapsed) + 0.5) or 0
        frameCount = 0
        fpsElapsed = 0
        statElapsed = 0

        local ping = getPing()

        pushSample(pingHistory, ping)
        pushSample(fpsHistory, fps)

        pingValueLabel.Text = ping .. "ms"
        fpsValueLabel.Text = fps .. " fps"

        redrawBars(pingBars, pingHistory)
        redrawBars(fpsBars, fpsHistory)
    end
end))
end

local dropdownRegistry = {}
local function CreateDropdown(parent, title, choices, defaultIndex, onChanged, yOffset)
    local dy = yOffset or 0
    local selected = defaultIndex or 1
    local choiceList = choices
    local open = false
    local setOpen
    local activeTween = nil
    local savedScroll = nil

    local button = Instance.new("TextButton")
    button.Size = UDim2.new(1, 0, 0, 28)
    button.LayoutOrder = #parent:GetChildren()
    button.BackgroundColor3 = Color3.fromRGB(26, 26, 26)
    button.BorderSizePixel = 0
    button.AutoButtonColor = false
    button.Text = ""
    button.Parent = parent

    local buttonCorner = Instance.new("UICorner")
    buttonCorner.CornerRadius = UDim.new(0, 6)
    buttonCorner.Parent = button

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -34, 1, 0)
    label.Position = UDim2.new(0, 12, 0, dy)
    label.BackgroundTransparency = 1
    label.Font = Enum.Font.GothamBold
    label.TextSize = 12
    label.TextColor3 = Color3.new(1, 1, 1)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Name = "DropdownLabel"
    label.Parent = button

    local chevron = Instance.new("TextLabel")
    chevron.Size = UDim2.new(0, 20, 1, 0)
    chevron.Position = UDim2.new(1, -26, 0, -2 + dy)
    chevron.BackgroundTransparency = 1
    chevron.Font = Enum.Font.GothamBold
    chevron.TextSize = 11
    chevron.TextColor3 = MUTED_TEXT
    chevron.Text = "v"
    chevron.Parent = button

    local rows = {}
    local function refreshRows()
        for i, row in ipairs(rows) do
            local isSelected = (i == selected)
            row.BackgroundColor3 = isSelected and Color3.fromRGB(48, 48, 48) or Color3.new(0, 0, 0)
            row.BackgroundTransparency = isSelected and 0 or 1
        end
        label.Text = title .. ": " .. (choiceList[selected] or "")
    end

    local dropdown = Instance.new("Frame")
    dropdown.BackgroundColor3 = Color3.fromRGB(22, 22, 22)
    dropdown.BorderSizePixel = 0
    dropdown.ClipsDescendants = true
    dropdown.Size = UDim2.new(1, 0, 0, 0)
    dropdown.LayoutOrder = button.LayoutOrder + 1
    dropdown.Visible = false
    table.insert(dropdownRegistry, dropdown)

    local dropdownCorner = Instance.new("UICorner")
    dropdownCorner.CornerRadius = UDim.new(0, 6)
    dropdownCorner.Parent = dropdown

    local dropdownLayout = Instance.new("UIListLayout")
    dropdownLayout.SortOrder = Enum.SortOrder.LayoutOrder
    dropdownLayout.Parent = dropdown

    local function buildRows()
        for _, child in ipairs(dropdown:GetChildren()) do
            if child:IsA("TextButton") then child:Destroy() end
        end
        for i in ipairs(rows) do rows[i] = nil end
        for i, choice in ipairs(choiceList) do
            local row = Instance.new("TextButton")
            row.Size = UDim2.new(1, 0, 0, 22)
            row.BackgroundColor3 = Color3.new(0, 0, 0)
            row.BackgroundTransparency = 1
            row.AutoButtonColor = false
            row.Font = Enum.Font.Gotham
            row.TextSize = 12
            row.TextColor3 = Color3.new(0.85, 0.85, 0.85)
            row.TextXAlignment = Enum.TextXAlignment.Left
            row.Text = "   " .. choice
            row.LayoutOrder = i - 1
            row.Parent = dropdown
            rows[i] = row
            row.MouseButton1Click:Connect(function()
                selected = i
                refreshRows()
                setOpen(false)
                if onChanged then
                    onChanged(choice, i)
                end
            end)
        end
        refreshRows()
    end
    buildRows()

    local function findScroller()
        local scroller = parent
        while scroller and not scroller:IsA("ScrollingFrame") do
            scroller = scroller.Parent
        end
        return scroller
    end
    setOpen = function(state)
        open = state
        chevron.Text = state and "^" or "v"
        if activeTween then
            activeTween:Cancel()
            activeTween = nil
        end
        if state then
            local scroller = findScroller()
            savedScroll = scroller and scroller.CanvasPosition or nil
            dropdown.Parent = parent
            dropdown.LayoutOrder = button.LayoutOrder + 1
            dropdown.Size = UDim2.new(1, 0, 0, 0)
            dropdown.Visible = true
            dropdown.BackgroundTransparency = 1
            for _, row in ipairs(rows) do
                row.BackgroundTransparency = 1
                row.TextTransparency = 1
            end
            activeTween = TweenService:Create(dropdown, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(1, 0, 0, #choiceList * 22),
                BackgroundTransparency = 0
            })
            local openTween = activeTween
            openTween:Play()
            for _, row in ipairs(rows) do
                local bgTarget = (row == rows[selected]) and 0 or 1
                TweenService:Create(row, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    BackgroundTransparency = bgTarget,
                    TextTransparency = 0
                }):Play()
            end
        else
            activeTween = TweenService:Create(dropdown, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
                Size = UDim2.new(1, 0, 0, 0)
            })
            local closeTween = activeTween
            closeTween:Play()
            closeTween.Completed:Connect(function()
                if not open then
                    dropdown.Visible = false
                    dropdown.Parent = nil
                    local node = parent
                    while node and node ~= screenGui do
                        for _, child in ipairs(node:GetChildren()) do
                            if child:IsA("UIListLayout") then
                                local order = child.SortOrder
                                child.SortOrder = Enum.SortOrder.Name
                                child.SortOrder = order
                            end
                        end
                        node = node.Parent
                    end
                    local scroller = findScroller()
                    if scroller and savedScroll then
                        local maxY = math.max(scroller.CanvasSize.Y.Offset - scroller.AbsoluteSize.Y, 0)
                        local pos = scroller.CanvasPosition
                        scroller.CanvasPosition = Vector2.new(pos.X, math.min(savedScroll.Y, maxY))
                    end
                    if scroller then
                        local p = scroller.CanvasPosition
                        scroller.CanvasPosition = Vector2.new(p.X, p.Y + 1)
                        task.defer(function()
                            if not open then
                                scroller.CanvasPosition = Vector2.new(p.X, p.Y)
                            end
                        end)
                    end
                end
                if activeTween == closeTween then
                    activeTween = nil
                end
            end)
        end
    end

    button.MouseButton1Click:Connect(function()
        setOpen(not open)
    end)
    local dropdownAPI
    dropdownAPI = {
        get = function() return selected end,
        set = function(i)
            i = math.clamp(math.floor((i or 1) + 0.5), 1, math.max(#choiceList, 1))
            selected = i
            refreshRows()
            if onChanged then
                onChanged(choiceList[i], i)
            end
        end,
        refresh = function(newList)
            choiceList = newList
            if selected > #choiceList then selected = math.max(#choiceList, 1) end
            buildRows()
        end,
    }
    controlAPIs["dropdown:" .. title] = dropdownAPI
    table.insert(configRegistry, { id = "dropdown:" .. title, get = dropdownAPI.get, set = dropdownAPI.set })

    task.defer(function()
        dropdown.Visible = true
        task.wait()
        if not open then
            dropdown.Visible = false
        end
    end)

    return button, dropdown
end

local noclipEnabled = false
local noclipConnection
local originalCollide = {}

local function applyNoclip(character)
    if not character then return end
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            if originalCollide[part] == nil then
                originalCollide[part] = part.CanCollide
            end
            part.CanCollide = false
        end
    end
end

local function restoreCollide()
    for part, wasCollidable in pairs(originalCollide) do
        if part and part.Parent then
            part.CanCollide = wasCollidable
        end
    end
    originalCollide = {}
end

local function setNoclip(state)
    noclipEnabled = state
    local character = LocalPlayer.Character
    if not character then return end

    if noclipConnection then
        noclipConnection:Disconnect()
        noclipConnection = nil
    end

    if state then
        applyNoclip(character)
        noclipConnection = RunService.Stepped:Connect(function()
            applyNoclip(character)
        end)
    else
        restoreCollide()
    end
end

LocalPlayer.CharacterAdded:Connect(function(character)
    originalCollide = {}
    if noclipEnabled then
        task.wait(0.5)
        setNoclip(true)
    end
end)

local noclipRow = CreateToggle(tabPanels["Misc"], "Noclip", false, setNoclip)
table.insert(featureRows, noclipRow)
tipFor(noclipRow, "Noclip", "Walk through walls and objects")
end

local function resolveKillerCharacterName(targetPlayer)
    local found = nil
    pcall(function()
        for _, val in pairs(targetPlayer:GetAttributes()) do
            if type(val) == "string" then
                if isKillerCharacter(val) then
                    found = val
                    return
                end
                local ch = KILLER_PERK_NAMES[val]
                if ch then
                    found = ch
                    return
                end
            end
        end
        local character = targetPlayer.Character
        if character then
            local first = character:GetAttribute("CosmeticFirstName")
            if isKillerCharacter(first) then
                found = first
                return
            end
            local perk = character:GetAttribute("Perk_Name")
            if isKillerCharacter(perk) then
                found = perk
                return
            end
            local ch = perk and KILLER_PERK_NAMES[perk]
            if ch then
                found = ch
                return
            end
            if isKillerCharacter(character.Name) then
                found = character.Name
            end
        end
    end)
    return found
end

function drawingBehindGui(x, y, w, h)
    local okVis, vis = pcall(function() return mainFrame.Visible end)
    if not okVis or not vis then return false end
    local okRect, px, py, sx, sy = pcall(function()
        local pos = mainFrame.AbsolutePosition
        local size = mainFrame.AbsoluteSize
        return pos.X, pos.Y, size.X, size.Y
    end)
    if not okRect or not px then return false end
    local x2 = px + sx
    local y2 = py + sy
    if w and h then
        return x < x2 and x + w > px and y < y2 and y + h > py
    end
    return x >= px and x <= x2 and y >= py and y <= y2
end

local function updateSurvivorBoxes()
    if not survivorESPEnabled or (survivorESPType ~= "Box" and survivorESPType ~= "Both") then
        for _, drawing in pairs(survivorBoxes) do
            drawing.Visible = false
        end
        return
    end
    local camera = workspace.CurrentCamera
    if not camera then return end
    local viewSize = camera.ViewportSize
    for player, drawing in pairs(survivorBoxes) do
        local visible = false
        if player ~= LocalPlayer and not isExcludedFromSurvivorESP(player) then
            local character = player.Character
            local rootPart = character and character:FindFirstChild("HumanoidRootPart")
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if rootPart and humanoid and humanoid.Health > 0 then
                local torso = character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso")
                local topPos = (torso and torso.Position or rootPart.Position) + Vector3.new(0, 4.8, 0)
                local bottomPos = rootPart.Position - Vector3.new(0, 4.8, 0)
                local top, topOnScreen = camera:WorldToViewportPoint(topPos)
                local bottom, botOnScreen = camera:WorldToViewportPoint(bottomPos)
                if topOnScreen or botOnScreen then
                    local screenSize = math.max(bottom.Y - top.Y, 10)
                    local boxWidth = screenSize * 0.65
                    if boxWidth < viewSize.X and screenSize < viewSize.Y then
                        local px = math.clamp(top.X - boxWidth / 2, 0, viewSize.X - boxWidth)
                        local py = math.clamp(top.Y, 0, viewSize.Y - screenSize)
                        px, py = math.floor(px + 0.5), math.floor(py + 0.5)
                        boxWidth, screenSize = math.floor(boxWidth + 0.5), math.floor(screenSize + 0.5)
                        drawing.Size = Vector2.new(boxWidth, screenSize)
                        drawing.Position = Vector2.new(px, py)
                        drawing.Color = Color3.fromRGB(50, 50, 255)
                        visible = not drawingBehindGui(px, py, boxWidth, screenSize)
                    end
                end
            end
        end
        drawing.Visible = visible
    end
end

trackLoop(RunService.RenderStepped:Connect(updateSurvivorBoxes))

        for _, drawing in pairs(killerBoxes) do
            drawing.Visible = false
        end
        return

    local viewSize = camera.ViewportSize
    for player, drawing in pairs(killerBoxes) do
        local visible = false
        if player ~= LocalPlayer and not isExcludedFromKillerESP(player) then
            local character = player.Character
            local rootPart = character and character:FindFirstChild("HumanoidRootPart")
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if rootPart and humanoid and humanoid.Health > 0 then
                local torso = character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso")
                local topPos = (torso and torso.Position or rootPart.Position) + Vector3.new(0, 4.8, 0)
                local bottomPos = rootPart.Position - Vector3.new(0, 4.8, 0)
                local top, topOnScreen = camera:WorldToViewportPoint(topPos)
                local bottom, botOnScreen = camera:WorldToViewportPoint(bottomPos)
                if topOnScreen or botOnScreen then
                    local screenSize = math.max(bottom.Y - top.Y, 10)
                    local boxWidth = screenSize * 0.65
                    if boxWidth < viewSize.X and screenSize < viewSize.Y then
                        local px = math.clamp(top.X - boxWidth / 2, 0, viewSize.X - boxWidth)
                        local py = math.clamp(top.Y, 0, viewSize.Y - screenSize)
                        px, py = math.floor(px + 0.5), math.floor(py + 0.5)
                        boxWidth, screenSize = math.floor(boxWidth + 0.5), math.floor(screenSize + 0.5)
                        drawing.Size = Vector2.new(boxWidth, screenSize)
                        drawing.Position = Vector2.new(px, py)
                        drawing.Color = Color3.fromRGB(255, 50, 50)
                        visible = not drawingBehindGui(px, py, boxWidth, screenSize)
                    end
                end
            end
        end
        drawing.Visible = visible
    end
end

local function getGeneratorProgress(generator)
    local progress = generator:GetAttribute("Progress")
    local needed = generator:GetAttribute("Needed")
    if typeof(progress) == "number" and typeof(needed) == "number" and needed > 0 then
        return math.floor((progress / needed) * 100)
    end
    local commonNames = {"Progress", "RepairProgress", "CurrentProgress", "Gauge", "Percent", "Value", "Amount", "Curr"}
    for _, child in pairs(generator:GetDescendants()) do
        if child:IsA("NumberValue") or child:IsA("IntValue") or child:IsA("FloatValue") then
            local nameLower = child.Name:lower()
            for _, pattern in ipairs(commonNames) do
                if string.find(nameLower, pattern:lower()) then
                    local val = child.Value
                    if type(val) == "number" then
                        if val <= 1 and val >= 0 then return math.floor(val * 100) end
                        return math.floor(math.clamp(val, 0, 100))
                    end
                end
            end
        end
    end
    for _, child in pairs(generator:GetDescendants()) do
        if child:IsA("NumberValue") then
            local val = child.Value
            if type(val) == "number" then
                if val <= 1 and val >= 0 then return math.floor(val * 100) end
                return math.floor(math.clamp(val, 0, 100))
            end
        end
    end
    return 0
end

    if not vaultESPEnabled then return end

    local camera = workspace.CurrentCamera

                    vaultLastFade[inst] = nil
                    if vaultNameDrawings[inst] then
                        removeVaultNameDrawing(inst)
                    end
                else
                    if showOutline then
                        if not highlight.Enabled then highlight.Enabled = true end
                        local fadeQ = math.floor(dist / vaultESPDistance * 20 + 0.5)
                        if vaultLastFade[inst] ~= fadeQ then
                            vaultLastFade[inst] = fadeQ
                            highlight.FillTransparency = fadeQ / 20 * 0.6
                            highlight.OutlineTransparency = fadeQ / 20 * 0.8
                        end

                        if highlight.Enabled then highlight.Enabled = false end
                        vaultLastFade[inst] = nil
                    end
                    if showText and camera then
                        local drawing = vaultNameDrawings[inst]
                        if not drawing then
                            local ok, d = pcall(Drawing.new, "Text")

                                trackDrawing(d)

local FLY_SPEED
do
local flying = false
local flyConnection
local flyOldPlatformStand = false
local flyOldWalkSpeed = 16
local flyOldGravity = 196.2
FLY_SPEED = 50

local function stopFly()
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    local character = LocalPlayer.Character
    if character then
        local root = character:FindFirstChild("HumanoidRootPart")
        if root then
            root.Velocity = Vector3.new(0, 0, 0)
            root.RotVelocity = Vector3.new(0, 0, 0)
        end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.PlatformStand = flyOldPlatformStand
            if humanoid.WalkSpeed == 0 then
                humanoid.WalkSpeed = flyOldWalkSpeed
            end
        end
    end
    workspace.Gravity = flyOldGravity
end

local function startFly()
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not hrp then return end

    stopFly()

    if humanoid then
        flyOldPlatformStand = humanoid.PlatformStand
        flyOldWalkSpeed = humanoid.WalkSpeed
        humanoid.PlatformStand = true
        humanoid.WalkSpeed = 0
    end
    flyOldGravity = workspace.Gravity
    workspace.Gravity = 0

    flyConnection = RunService.RenderStepped:Connect(function()
        if not hrp.Parent then return end
        if UserInputService:GetFocusedTextBox() then return end
        local camera = workspace.CurrentCamera
        if not camera then return end
        hrp.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + camera.CFrame.LookVector)
        local cf = camera.CFrame
        local moveVector = Vector3.new(0, 0, 0)

        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVector += cf.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVector -= cf.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVector -= cf.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVector += cf.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveVector += Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveVector -= Vector3.new(0, 1, 0) end

        if moveVector.Magnitude > 0 then
            moveVector = moveVector.Unit * FLY_SPEED
        end

        hrp.Velocity = moveVector
        hrp.RotVelocity = Vector3.new(0, 0, 0)
    end)
end

local function setFly(state)
    flying = state
    if state then
        startFly()
    else
        stopFly()
    end
end

LocalPlayer.CharacterAdded:Connect(function()
    if flying then
        for _, delay in ipairs({0.5, 2, 5}) do
            task.delay(delay, function()
                if flying and LocalPlayer.Character then
                    setFly(true)
                end
            end)
        end
    end
end)
do
    local flyKeepNext = 0
    trackLoop(RunService.Heartbeat:Connect(function()
        if not flying then return end
        local now = os.clock()
        if now < flyKeepNext then return end
        flyKeepNext = now + 2
        local character = LocalPlayer.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if character and humanoid and root and not humanoid.PlatformStand then
            setFly(true)
        elseif flying and not flyConnection and character and root then
            setFly(true)
        end
    end))
end

        local value = math.clamp(math.floor(rawValue / sliderStep + 0.5) * sliderStep, minVal, maxVal)
        currentVal = value

        fill.Size = UDim2.new(0, ratio * SLIDER_TRACK_WIDTH, 1, 0)
        knob.Position = UDim2.new(0, ratio * SLIDER_TRACK_WIDTH, 0.5, 0)
        valueLabel.Text = value .. "/" .. maxVal
        if fire then
            pcall(function() mainFrame:SetAttribute("AeonSliderMovedAt", os.clock()) end)
            if onChanged then
                onChanged(value)
            end
        end
        return value
    end

    local sliderAPI = { get = function() return currentVal end, set = function(v) setValue(v, true) end }
    controlAPIs["slider:" .. labelText] = sliderAPI
    table.insert(configRegistry, { id = "slider:" .. labelText, get = sliderAPI.get, set = sliderAPI.set })

    local dragging = false
    local function updateFromInput(input)
        local relativeX = input.Position.X - track.AbsolutePosition.X
        local ratio = math.clamp(relativeX / SLIDER_TRACK_WIDTH, 0, 1)
        setValue(minVal + ratio * (maxVal - minVal), true)
    end

    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromInput(input)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromInput(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    return row
end

local function stripLowgfxInstance(part)
    if part:IsA("BasePart") then
        if savedParts[part] == nil then
            savedParts[part] = { part.Material, part.Reflectance }
        end
        part.Material = Enum.Material.SmoothPlastic
        part.Reflectance = 0
    elseif part:IsA("Texture") or part:IsA("Decal") then
        if savedParts[part] == nil then
            savedParts[part] = { nil, part:Clone(), part.Parent }
        end
        part:Destroy()
    elseif part:IsA("SpecialMesh") then
        if savedParts[part] == nil then
            savedParts[part] = { "mesh", "TextureId", part.TextureId }
        end
        part.TextureId = ""
    elseif part:IsA("MeshPart") then
        if savedParts[part] == nil then
            savedParts[part] = { "mesh", "TextureID", part.TextureID }
        end
        part.TextureID = ""
    end
end

local function hookLowgfx()
    if lowgfxConn then
        lowgfxConn:Disconnect()
        lowgfxConn = nil
    end
    lowgfxConn = trackLoop(workspace.DescendantAdded:Connect(function(inst)
        if lowgfxEnabled then
            stripLowgfxInstance(inst)
        end
    end))
end

local function applyLowgfx()
    local lighting = game:GetService("Lighting")
    if lowgfxEnabled then
        if savedLight.GlobalShadows == nil then
            savedLight.GlobalShadows = lighting.GlobalShadows
            savedLight.FogEnd = lighting.FogEnd
        end
        lighting.GlobalShadows = false
        lighting.FogEnd = 999999
        for _, child in ipairs(lighting:GetChildren()) do
            if child:IsA("PostEffect") then
                if savedEffects[child] == nil then
                    savedEffects[child] = child.Enabled
                end
                child.Enabled = false
            end
        end

        for _, part in ipairs(game:GetService("Workspace"):GetDescendants()) do
            stripLowgfxInstance(part)
        end
        hookLowgfx()
    else
        if lowgfxConn then
            lowgfxConn:Disconnect()
            lowgfxConn = nil
        end
        if savedLight.GlobalShadows ~= nil then
            lighting.GlobalShadows = savedLight.GlobalShadows
            lighting.FogEnd = savedLight.FogEnd
        end
        for inst, wasEnabled in pairs(savedEffects) do
            if inst and inst.Parent then
                inst.Enabled = wasEnabled
            end
        end
        for part, saved in pairs(savedParts) do
            if saved[1] == "mesh" then
                if part and part.Parent then
                    part[saved[2]] = saved[3]
                end
                savedParts[part] = nil
            elseif saved[1] ~= nil then
                if part and part.Parent then
                    part.Material = saved[1]
                    part.Reflectance = saved[2]
                end
            else
                local clone, parent = saved[2], saved[3]
                if parent then
                    clone.Parent = parent
                end
                savedParts[part] = nil
            end
        end
    end
end

local function setLowgfx(state)
    lowgfxEnabled = state
    applyLowgfx()
end

local CONFIG_FOLDER = "AeonHubDB2Configs"
local function configPath(name)
    return CONFIG_FOLDER .. "/" .. name .. ".txt"
end
local function sanitizeConfigName(name)
    name = string.match(name or "", "^%s*(.-)%s*$")
    name = string.gsub(name, "[^%w_%- ]", "")
    name = string.gsub(name, "%s+", " ")
    return name
end
local function readConfigIndex()
    local names = {}
    local scanned = nil
    pcall(function()
        if listfiles then
            scanned = {}
            for _, f in ipairs(listfiles(CONFIG_FOLDER)) do
                local base = string.match(f, "([^/\\]+)%.txt$")
                if base and base ~= "index" then
                    local nm = sanitizeConfigName(base)
                    if nm ~= "" then table.insert(scanned, nm) end
                end
            end
        end
    end)
    if scanned then
        pcall(function()
            writefile(CONFIG_FOLDER .. "/index.txt", table.concat(scanned, "\n"))
        end)
        return scanned
    end
    pcall(function()
        local content = readfile(CONFIG_FOLDER .. "/index.txt")
        if content then
            for line in string.gmatch(content, "[^\r\n]+") do
                local nm = sanitizeConfigName(line)
                if nm ~= "" then table.insert(names, nm) end
            end
        end
    end)
    return names
end
local function writeConfigIndex(names)
    pcall(function()
        writefile(CONFIG_FOLDER .. "/index.txt", table.concat(names, "\n"))
    end)
end
local function saveConfig(name)
    local lines = {}
    for _, c in ipairs(configRegistry) do
        if c.id ~= "dropdown:Config theme" then
            local ok, val = pcall(c.get)
            if ok and val ~= nil then table.insert(lines, c.id .. "=" .. tostring(val)) end
        end
    end
    pcall(function()
        writefile(configPath(name), table.concat(lines, "\n"))
    end)
end
local function loadConfig(name)
    local content = nil
    pcall(function()
        content = readfile(configPath(name))
    end)
    if not content then return end
    local map = {}
    for line in string.gmatch(content, "[^\r\n]+") do
        local id, val = string.match(line, "^([^=]+)=(.+)$")
        if id and val then map[id] = val end
    end
    for _, c in ipairs(configRegistry) do
        local val = map[c.id]
        if val ~= nil and c.id ~= "dropdown:Config theme" then
            if string.sub(c.id, 1, 7) == "toggle:" then
                pcall(c.set, val == "true")
            elseif string.sub(c.id, 1, 7) == "slider:" then
                pcall(c.set, tonumber(val) or 0)
            elseif string.sub(c.id, 1, 8) == "dropdown:" then
                pcall(c.set, tonumber(val) or 1)
            elseif string.sub(c.id, 1, 8) == "keybind:" then
                pcall(c.set, val)
            elseif string.sub(c.id, 1, 6) == "color:" then
                pcall(c.set, val)
            else
                pcall(c.set, tonumber(val) or 1)
            end
        end
    end
end

local function isLocalSurvivor()

local function setupStaminaHold()
    if staminaSignalConn then
        staminaSignalConn:Disconnect()
        staminaSignalConn = nil
    end

    local handler = character and character:FindFirstChild("SurvivorHandlerServer")
    if not handler then

    end

        if staminaEnabled and isLocalSurvivor() then

local function animSpeedFor(aid, hasPerk)
    local s = animSpeeds[aid]

local function hasStealthyPerk()
    local ok, hudPerk = pcall(function()
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        local hud = pg and pg:FindFirstChild("HUD")
        local info = hud and hud:FindFirstChild("Char_Info")
        local ab = info and info:FindFirstChild("Ability_Survivor")
        local lbl = ab and ab:FindFirstChild("PerkName")
        return lbl and lbl.Text or nil
    end)
    if ok and type(hudPerk) == "string" and string.find(hudPerk, "Stealthy") then
        return true
    end
    local okMenu, menuPerk = pcall(function()
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        local menus = pg and pg:FindFirstChild("Menus")
        local pre = menus and menus:FindFirstChild("Pregame_Menu")
        local perks = pre and pre:FindFirstChild("Perks")
        local disp = perks and perks:FindFirstChild("DisplayText")
        return disp and disp.Text or nil
    end)
    if okMenu and type(menuPerk) == "string" and string.find(menuPerk, "Stealthy") then
        return true
    end
    for _, v in pairs(LocalPlayer:GetAttributes()) do
        if v == "Stealthy" then return true end
    end
    local character = LocalPlayer.Character
    if character then
        for _, v in pairs(character:GetAttributes()) do
            if v == "Stealthy" then return true end
        end
        for _, d in ipairs(character:GetDescendants()) do
            if d:IsA("StringValue") or d:IsA("BoolValue") then
                if d.Value == "Stealthy" or d.Name == "Stealthy" then
                    return true
                end
            end
        end
    end
    return false
end
local function hookAnimSpeeds(character)
    task.spawn(function()
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if not humanoid and character then
            local ok, child = pcall(function() return character:WaitForChild("Humanoid", 5) end)
            if ok then humanoid = child end
        end
        if not humanoid then return end
        humanoid.AnimationPlayed:Connect(function(track)
            local anim = track.Animation
            local id = anim and tostring(anim.AnimationId or "")
            local matched = nil
            for animId in pairs(animSpeeds) do
                if string.find(id, animId, 1, true) then
                    matched = animId
                    break
                end
            end
            if matched then
                animActive[matched] = tick()
                if staminaEnabled then
                    humanoid.WalkSpeed = animSpeedFor(matched, hasStealthyPerk())
                end
                track.Stopped:Connect(function()
                    animActive[matched] = nil
                    if staminaEnabled then
                        local latestSpeed = nil
                        local latestTime = 0
                        for aid, t in pairs(animActive) do
                            local s = animSpeedFor(aid, hasStealthyPerk())
                            if s and t > latestTime then
                                latestTime = t
                                latestSpeed = s
                            end
                        end
                        if latestSpeed then
                            humanoid.WalkSpeed = latestSpeed
                        else
                            local whoNow = resolveMyCharacter()
                            local baseNow = whoNow and (14.625 + 0.125 * SURVIVOR_SPEED[whoNow]) or 15.5
                            humanoid.WalkSpeed = baseNow
                        end
                    end
                end)
            end
        end)
    end)
end

resolveMyCharacter = function()

            if SURVIVOR_SPEED[v] then return v end

            if c then return c end
        end
        return nil

        local found = check(v)
        if found then return found end
    end

            if found then return found end
        end
    end

local function killerNameFromValue(v)
    if type(v) ~= "string" then return nil end
    if KILLER_CHARACTER_NAMES[v] then return v end

local matchKillerName = nil
local killerCheckAt = 0
local function refreshMatchKiller()
    local nowT = os.clock()
    if nowT < killerCheckAt then return end
    killerCheckAt = nowT + 5
    local foundName = nil
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local team = p.Team
            local tname = team and string.lower(team.Name) or ""
            if string.find(tname, "kill") then
                local ch = p.Character
                local fn = ch and ch:GetAttribute("CosmeticFirstName")
                foundName = killerNameFromValue(fn)
                if not foundName then
                    for _, v in pairs(p:GetAttributes()) do
                        foundName = killerNameFromValue(v)
                        if foundName then break end
                    end
                end
                if not foundName and ch then
                    for _, v in pairs(ch:GetAttributes()) do
                        if v ~= fn then
                            foundName = killerNameFromValue(v)
                            if foundName then break end
                        end
                    end
                end
                break
            end
        end
    end
    matchKillerName = foundName
end

local trackBarsOn = false
local trackBars = {}
local trackEst = {}
local killerStamActive = false
local function removeTrackBar(targetPlayer)
    local set = trackBars[targetPlayer]
    if set then
        if set.bg then pcall(function() set.bg:Destroy() end) end
        if set.fill then pcall(function() set.fill:Destroy() end) end
        trackBars[targetPlayer] = nil
    end
end
local function liveStam(targetPlayer)
    local ch = targetPlayer.Character
    local handler = ch and ch:FindFirstChild("SurvivorHandlerServer")
    if not handler then
        local wc = workspace:FindFirstChild(targetPlayer.Name)
        handler = wc and wc:FindFirstChild("SurvivorHandlerServer")
    end
    if not handler then return nil end
    local cur, max
    pcall(function()
        cur = handler:GetAttribute("CurrentStamina")
        max = handler:GetAttribute("MaxStamina")
    end)
    if type(cur) ~= "number" or type(max) ~= "number" or max <= 0 then return nil end
    return math.clamp(cur / max, 0, 1)
end
local function trackPct(targetPlayer, dt)
    local ch = targetPlayer.Character
    if not ch then trackEst[targetPlayer] = nil return nil end
    local handler = ch:FindFirstChild("SurvivorHandlerServer")
    if not handler then
        local wc = workspace:FindFirstChild(targetPlayer.Name)
        handler = wc and wc:FindFirstChild("SurvivorHandlerServer")
    end
    if not handler then trackEst[targetPlayer] = nil return nil end
    local ok, live = pcall(liveStam, targetPlayer)
    if not ok then live = nil end
    if type(live) ~= "number" then live = nil end
    local hum = ch:FindFirstChildOfClass("Humanoid")
    local est = trackEst[targetPlayer]
    if est == nil then est = 1 end
    if live and live > 0.01 then
        est = est + (live - est) * math.min(1, dt * 10)
    elseif hum then
        if hum.WalkSpeed > 18 then
            est = est - dt / 5
        else
            est = est + dt / 9
        end
        est = math.clamp(est, 0, 1)
    end
    trackEst[targetPlayer] = est
    return est
end
local function ensureTrackBar(targetPlayer)
    local set = trackBars[targetPlayer]
    if set and set.bg and set.bg.Parent and set.fill and set.fill.Parent then return set end

local function setTrackStamina(state)
    trackBarsOn = state
    if not state then
        for p in pairs(trackBars) do removeTrackBar(p) end
        trackEst = {}
    end
end
local trackLast = 0
trackLoop(RunService.Heartbeat:Connect(function()
    if not trackBarsOn then return end
    if not isLocalKiller() then
        if killerStamActive then
            killerStamActive = false
            for p in pairs(trackBars) do removeTrackBar(p) end
            trackEst = {}
        end
        return
    end
    killerStamActive = true
    local now = os.clock()
    local dt = math.min(now - trackLast, 0.1)
    trackLast = now
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    local camera = workspace.CurrentCamera
    if not (myRoot and camera) then return end
    local viewSize = camera.ViewportSize
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local pct = trackPct(p, dt)
            local set = pct and ensureTrackBar(p) or nil
            if set and pct then
                local ch = p.Character
                local root = ch and ch:FindFirstChild("HumanoidRootPart")
                if root then
                    local dist = (root.Position - myRoot.Position).Magnitude
                    if dist <= 70 and dist > 0.5 then
                        local torso = ch:FindFirstChild("Torso") or ch:FindFirstChild("UpperTorso")
                        local topPos = (torso and torso.Position or root.Position) + Vector3.new(0, 4.8, 0)
                        local bottomPos = root.Position - Vector3.new(0, 4.8, 0)
                        local top, topOn = camera:WorldToViewportPoint(topPos)
                        local bottom, botOn = camera:WorldToViewportPoint(bottomPos)
                        if topOn or botOn then
                            local screenSize = math.max(bottom.Y - top.Y, 10)
                            local effH = math.min(screenSize, viewSize.Y * 0.9)
                            local effW = effH * 0.65
                            local boxX = math.max(math.min(top.X - effW / 2, viewSize.X - effW - 14), 0)
                            local boxY = math.max(math.min(top.Y, viewSize.Y - effH - 4), 0)
                            local bw = 60
                            local bx = boxX + effW / 2 - bw / 2
                            local by = boxY + effH + 4
                            local vis = not drawingBehindGui(bx, by, bw, 8)
                            set.bg.Position = UDim2.new(0, bx, 0, by)
                            set.bg.Visible = vis
                            set.fill.Size = UDim2.new(0, math.max((bw - 2) * pct, 0), 0, 6)
                        else
                            set.bg.Visible = false
                        end
                    else
                        set.bg.Visible = false
                    end
                else
                    set.bg.Visible = false
                end
            elseif set then
                set.bg.Visible = false
            end
            if not p.Character then removeTrackBar(p) end
        end
    end
    for p in pairs(trackBars) do
        if not p.Parent or p == LocalPlayer then removeTrackBar(p) end
    end
end))

    if killerLightSaved ~= nil then
        local light = findKillerLight()
        if light then
            pcall(function() light.Color = killerLightSaved end)
        end
        killerLightSaved = nil
    end

    killerLightEnabled = state

local function killerLightHex(c)
    return string.format("%02X%02X%02X",
        math.floor(c.R * 255 + 0.5),
        math.floor(c.G * 255 + 0.5),
        math.floor(c.B * 255 + 0.5))
end
local function killerLightFromHex(h)
    if type(h) ~= "string" then return nil end
    h = h:gsub("#", "")
    if #h ~= 6 then return nil end
    local r = tonumber(h:sub(1, 2), 16)
    local g = tonumber(h:sub(3, 4), 16)
    local b = tonumber(h:sub(5, 6), 16)
    if not (r and g and b) then return nil end
    return Color3.fromRGB(r, g, b)
end

    local function hueFromX(px)
        local w = bar.AbsoluteSize.X
        if w <= 0 then return 0 end
        return math.clamp((px - bar.AbsolutePosition.X) / w, 0, 1)
    end

    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            barDragging = true
            local c = Color3.fromHSV(hueFromX(input.Position.X), 1, 1)
            pickLightColor(c)
            refreshPickerUI(c)
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then barDragging = false end
            end)
        end
    end)
    bar.InputChanged:Connect(function(input)
        if barDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local c = Color3.fromHSV(hueFromX(input.Position.X), 1, 1)
            pickLightColor(c)
            refreshPickerUI(c)
        end
    end)
    refreshPickerUI(KILLER_LIGHT_CUSTOM)
    TweenService:Create(popup, TweenInfo.new(0.22, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, 200, 0, 118),
        GroupTransparency = 0
    }):Play()
end

do
    local clickStart
    local guiVisible = false
    local SMALL_SIZE = UDim2.new(0, 0, 0, 0)

    local function buttonCenter()
        return UDim2.new(
            toggleButton.Position.X.Scale, toggleButton.Position.X.Offset + 25,
            toggleButton.Position.Y.Scale, toggleButton.Position.Y.Offset + 25
        )
    end

    toggleButton.MouseButton1Down:Connect(function()
        clickStart = { x = toggleButton.Position.X.Offset, y = toggleButton.Position.Y.Offset }
    end)

    toggleButton.MouseButton1Up:Connect(function()
        if not clickStart then return end
        local pos = toggleButton.Position
        local moved = math.abs(pos.X.Offset - clickStart.x) + math.abs(pos.Y.Offset - clickStart.y)
        clickStart = nil
        if moved >= 10 then return end

        guiVisible = not guiVisible
        if guiVisible then
            mainFrame.Visible = true
            contentArea.Visible = false
            mainFrame.Size = SMALL_SIZE
            mainFrame.Position = buttonCenter()
            for _, dd in ipairs(dropdownRegistry) do dd.Visible = true end
            task.wait()
            for _, dd in ipairs(dropdownRegistry) do dd.Visible = false end
            do
                local bi = (selectedBottomTab == "Options") and 2 or 1
                local estimateX = (bi - 0.5) * 0.5 * FULL_SIZE.X.Offset - 10
                TweenService:Create(bottomUnderlineTop, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Position = UDim2.new(0, estimateX, 0, -6)
                }):Play()
            end
            local openTween = TweenService:Create(mainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = FULL_SIZE,
                Position = SAVED_POS
            })
            openTween:Play()
            openTween.Completed:Connect(function()
                if guiVisible then
                    task.spawn(function()
                        for _, panel in pairs(tabPanels) do
                            panel.Visible = false
                        end
                        task.wait()
                        if not guiVisible then return end
                        for panelName, panel in pairs(tabPanels) do
                            panel.Visible = (panelName == currentTab)
                        end
                        contentArea.Visible = true
                        selectTab(currentTab, false)
                        selectBottomTab(selectedBottomTab, false)
                    end)
                end
            end)
        else
            local closeTween = TweenService:Create(mainFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
                Size = SMALL_SIZE,
                Position = buttonCenter()
            })
            closeTween:Play()
            closeTween.Completed:Connect(function()
                if not guiVisible then
                    mainFrame.Visible = false
                end
            end)
        end
    end)

    do
        local function pokeAll()
            for _, panel in pairs(tabPanels) do
                for _, child in ipairs(panel:GetDescendants()) do
                    if child:IsA("UIListLayout") then
                        local order = child.SortOrder
                        child.SortOrder = Enum.SortOrder.Name
                        child.SortOrder = order
                    end
                end
            end
        end
        task.delay(2, pokeAll)
        task.delay(5, pokeAll)
    end

    local pressStartTime
    toggleButton.MouseButton1Down:Connect(function()
        pressStartTime = os.clock()
    end)
    toggleButton.MouseButton1Up:Connect(function()
        if pressStartTime and os.clock() - pressStartTime < 0.25 then
            local p = toggleButton.Position
            local shrink = TweenService:Create(toggleButton, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 44, 0, 44),
                Position = UDim2.new(p.X.Scale, p.X.Offset + 3, p.Y.Scale, p.Y.Offset + 3)
            })
            shrink:Play()
            TweenService:Create(logo, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 30, 0, 30),
                TextSize = 26
            }):Play()
            shrink.Completed:Connect(function()
                local p2 = toggleButton.Position
                TweenService:Create(toggleButton, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = BUTTON_FULL_SIZE,
                    Position = UDim2.new(p2.X.Scale, p2.X.Offset - 3, p2.Y.Scale, p2.Y.Offset - 3)
                }):Play()
                TweenService:Create(logo, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = LOGO_FULL_SIZE,
                    TextSize = 34
                }):Play()
            end)
        end
        pressStartTime = nil
    end)
